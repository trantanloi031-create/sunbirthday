    <script>

        // --- Vui lòng chỉnh sửa các biến sau ---

        const eventDateString = "Nov 25, 2023 18:00:00"; // Định dạng: Thg Ngày, Năm HH:MM:SS

        const sunsPhotoUrl = "https://via.placeholder.com/150/FFD700/FFFFFF?text=Sun"; // Thay bằng URL ảnh của Sun

        const googleMapsLink = "https://maps.app.goo.gl/YourGoogleMapsLinkHere"; // Thay bằng link Google Maps thực tế

        const rsvpFormLink = "https://forms.gle/YourRSVPFormLinkHere"; // Thay bằng link form xác nhận tham gia

        const musicUrl = "https://www.bensound.com/bensound-music/bensound-happyrock.mp3"; // Thay bằng URL nhạc nền khác nếu muốn


        // --- Cập nhật HTML với các biến tùy chỉnh ---

        document.getElementById('sunPhoto').src = sunsPhotoUrl;

        document.querySelector('.event-details a').href = googleMapsLink;

        document.querySelector('.rsvp-button').href = rsvpFormLink;

        document.getElementById('backgroundMusic').src = musicUrl;



        // --- Bộ đếm ngược (Countdown Timer) ---

        const countdownElement = document.getElementById('countdown');

        const eventDateTime = new Date(eventDateString).getTime();


        const updateCountdown = setInterval(function() {

            const now = new Date().getTime();

            const distance = eventDateTime - now;


            const days = Math.floor(distance / (1000 * 60 * 60 * 24));

            const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));

            const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));

            const seconds = Math.floor((distance % (1000 * 60)) / 1000);


            if (distance < 0) {

                clearInterval(updateCountdown);

                countdownElement.innerHTML = "🎉 TIỆC ĐANG DIỄN RA! CHÚC MỪNG SINH NHẬT SUN! 🎉";

            } else {

                countdownElement.innerHTML = `Chỉ còn: ${days} ngày ${hours} giờ ${minutes} phút ${seconds} giây!`;

            }

        }, 1000);


        // --- Bật/Tắt Nhạc ---

        const backgroundMusic = document.getElementById('backgroundMusic');

        const toggleMusicButton = document.getElementById('toggleMusicButton');

        let isPlaying = true; // Giả sử nhạc tự động phát


        toggleMusicButton.addEventListener('click', function() {

            if (isPlaying) {

                backgroundMusic.pause();

                toggleMusicButton.textContent = "Bật Nhạc";

            } else {

                backgroundMusic.play();

                toggleMusicButton.textContent = "Tắt Nhạc";

            }

            isPlaying = !isPlaying;

        });


        // --- Hiệu ứng Lấp lánh (Sparkle Effect) ---

        const sparkleContainer = document.getElementById('sparkle-container');


        function createSparkle() {

            const sparkle = document.createElement('div');

            sparkle.classList.add('sparkle');

            sparkle.style.width = Math.random() * 8 + 4 + 'px'; 

            sparkle.style.height = sparkle.style.width;

            sparkle.style.left = Math.random() * 100 + 'vw'; 

            sparkle.style.animationDelay = Math.random() * 3 + 's'; 

            sparkle.style.animationDuration = Math.random() * 2 + 3 + 's'; 

            sparkleContainer.appendChild(sparkle);


            // Xóa phần tử sparkle sau khi hoàn tất animation

            sparkle.addEventListener('animationend', () => {

                sparkle.remove();

            });

        }


        // Khởi tạo các hạt lấp lánh ban đầu và tạo liên tục

        setInterval(createSparkle, 200); 

        for (let i = 0; i < 50; i++) { 

            createSparkle();

        }
